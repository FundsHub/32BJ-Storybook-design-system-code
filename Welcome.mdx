export const WELCOME_MOTION_VERSION = 'V4-load-parallax-hover-drift-squircle';

export const WelcomeMotion = ({ children }) => {
  const rootRef = useRef(null);

  useLayoutEffect(() => {
    const root = rootRef.current;
    if (!root) return;

    const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
    const cards = Array.from(root.querySelectorAll('.ds-welcome-card'));
    const sectionLayers = Array.from(
      root.querySelectorAll('.ds-welcome-metrics, .ds-welcome-section, .ds-welcome-close')
    );
    const settleTimers = new Set();
    let forceMotion = false;
    let frameId = 0;
    let loadFrameOne = 0;
    let loadFrameTwo = 0;
    let loadTimer = 0;
    let cardObserver = null;

    try {
      forceMotion = new URLSearchParams(window.parent.location.search).get('motion') === 'on';
    } catch {
      forceMotion = new URLSearchParams(window.location.search).get('motion') === 'on';
    }

    if (forceMotion) root.classList.add('ds-welcome-force-motion');

    const motionDisabled = () => reduceMotion.matches && !forceMotion;
    const clamp = (value, minimum, maximum) => Math.min(Math.max(value, minimum), maximum);

    const updateParallax = () => {
      frameId = 0;

      if (motionDisabled()) {
        root.style.removeProperty('--ds-parallax-far');
        root.style.removeProperty('--ds-parallax-mid');
        root.style.removeProperty('--ds-parallax-accent');
        sectionLayers.forEach((section) => section.style.removeProperty('--ds-section-shift'));
        return;
      }

      const distance = clamp(window.scrollY, 0, 620);
      root.style.setProperty('--ds-parallax-far', `${distance * 0.07}px`);
      root.style.setProperty('--ds-parallax-mid', `${distance * 0.12}px`);
      root.style.setProperty('--ds-parallax-accent', `${distance * -0.055}px`);

      const viewportCenter = window.innerHeight / 2;
      sectionLayers.forEach((section, index) => {
        const bounds = section.getBoundingClientRect();
        const sectionCenter = bounds.top + bounds.height / 2;
        const depth = index % 2 === 0 ? 0.065 : 0.045;
        const shift = clamp((viewportCenter - sectionCenter) * depth, -46, 46);
        section.style.setProperty('--ds-section-shift', `${shift}px`);
      });
    };

    const requestUpdate = () => {
      if (!frameId) frameId = window.requestAnimationFrame(updateParallax);
    };

    const configureCardReveals = () => {
      cardObserver?.disconnect();
      settleTimers.forEach((timer) => window.clearTimeout(timer));
      settleTimers.clear();

      if (motionDisabled() || !('IntersectionObserver' in window)) {
        root.classList.remove('ds-welcome-motion-ready');
        root.classList.add('ds-welcome-loaded');
        cards.forEach((card) => {
          card.removeAttribute('data-reveal');
          card.style.removeProperty('--ds-card-delay');
        });
        return;
      }

      cards.forEach((card) => {
        const parent = card.parentElement;
        const siblings = parent
          ? Array.from(parent.children).filter((item) => item.classList.contains('ds-welcome-card'))
          : cards;
        const itemIndex = Math.max(siblings.indexOf(card), 0);
        const columns = parent?.classList.contains('ds-welcome-categories')
          ? 2
          : parent?.classList.contains('ds-welcome-patterns') ||
              parent?.classList.contains('ds-welcome-audiences')
            ? 3
            : 4;

        card.setAttribute('data-reveal', 'pending');
        card.style.setProperty('--ds-card-delay', `${(itemIndex % columns) * 120}ms`);
      });

      root.classList.add('ds-welcome-motion-ready');
      cardObserver = new IntersectionObserver(
        (entries) => {
          entries.forEach((entry) => {
            if (!entry.isIntersecting) return;
            const card = entry.target;
            const delay = Number.parseInt(card.style.getPropertyValue('--ds-card-delay'), 10) || 0;
            card.setAttribute('data-reveal', 'visible');
            cardObserver?.unobserve(entry.target);

            const timer = window.setTimeout(() => {
              card.setAttribute('data-reveal', 'settled');
              card.style.removeProperty('--ds-card-delay');
              settleTimers.delete(timer);
            }, 880 + delay);
            settleTimers.add(timer);
          });
        },
        { rootMargin: '0px 0px -14% 0px', threshold: 0.18 }
      );

      cards.forEach((card) => cardObserver.observe(card));
    };

    const startLoadAnimation = () => {
      if (loadFrameOne) window.cancelAnimationFrame(loadFrameOne);
      if (loadFrameTwo) window.cancelAnimationFrame(loadFrameTwo);
      if (loadTimer) window.clearTimeout(loadTimer);

      root.classList.remove('ds-welcome-loaded');
      if (motionDisabled()) {
        root.classList.add('ds-welcome-loaded');
        return;
      }

      loadFrameOne = window.requestAnimationFrame(() => {
        loadFrameTwo = window.requestAnimationFrame(() => {
          loadTimer = window.setTimeout(() => {
            root.classList.add('ds-welcome-loaded');
          }, 220);
        });
      });
    };

    const handleMotionChange = () => {
      configureCardReveals();
      startLoadAnimation();
      requestUpdate();
    };

    configureCardReveals();
    startLoadAnimation();
    updateParallax();
    window.addEventListener('scroll', requestUpdate, { passive: true });
    window.addEventListener('resize', requestUpdate);
    reduceMotion.addEventListener('change', handleMotionChange);

    return () => {
      if (frameId) window.cancelAnimationFrame(frameId);
      if (loadFrameOne) window.cancelAnimationFrame(loadFrameOne);
      if (loadFrameTwo) window.cancelAnimationFrame(loadFrameTwo);
      if (loadTimer) window.clearTimeout(loadTimer);
      cardObserver?.disconnect();
      settleTimers.forEach((timer) => window.clearTimeout(timer));
      window.removeEventListener('scroll', requestUpdate);
      window.removeEventListener('resize', requestUpdate);
      reduceMotion.removeEventListener('change', handleMotionChange);
    };
  }, []);

  return <div className="ds-welcome-motion-root" ref={rootRef}>{children}</div>;
};
